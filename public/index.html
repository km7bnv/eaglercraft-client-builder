// FINAL STANDALONE builder.js - Fetches /base.html → MODS COPY → DOWNLOADS NEW FILE // BASE FILE STAYS 100% UNCHANGED! Modded version downloads separately // (function() { document.addEventListener('DOMContentLoaded', init); function init() { bindInputs(); updatePreview(); window.buildClient = buildClient; } function bindInputs() { const ids = ['clientName','username','seed','bgColor','motd','splashText','texturePack','fly','fullbright','xray','nofall','speed','killaura','killauraRange']; ids.forEach(id => document.getElementById(id)?.addEventListener('input', updatePreview)); document.getElementById('addServer')?.addEventListener('click', addServer); } function updatePreview() { const name = document.getElementById('clientName')?.value || 'Modded Client'; const hacks = getEnabledHacks(); const serverCount = document.querySelectorAll('.server-entry').length; const previewEl = document.getElementById('preview'); if (previewEl) { previewEl.textContent = `${name} | Hacks: ${hacks.join(', ') || 'None'} | Servers: ${serverCount}`; } } function getEnabledHacks() { return ['fly', 'fullbright', 'xray', 'nofall', 'speed', 'killaura'] .filter(id => document.getElementById(id)?.checked) .map(id => id.charAt(0).toUpperCase() + id.slice(1)); } function addServer() { const container = document.getElementById('serverList'); const entry = document.createElement('div'); entry.className = 'server-entry'; entry.innerHTML = ` `; container.appendChild(entry); } // 🔥 MAIN FUNCTION - BASE UNTOUCHED, DOWNLOADS MODDED COPY async function buildClient() { const output = document.getElementById('output'); output.textContent = '📥 Fetching base.html → Modifying COPY → Downloading...'; try { // 1. COLLECT SETTINGS const clientName = (document.getElementById('clientName').value || 'ModdedEagler').replace(/[^a-zA-Z0-9_-]/g, ''); const settings = { title: clientName, username: document.getElementById('username').value || 'Player', seed: parseInt(document.getElementById('seed').value) || 12345, bgColor: document.getElementById('bgColor').value || '#1a1a1a', motd: document.getElementById('motd').value || `${clientName} Client`, splash: document.getElementById('splashText').value || 'Hacks Enabled!', texturePack: document.getElementById('texturePack').value || 'default', servers: Array.from(document.querySelectorAll('.server-entry')).map(row => { const inputs = row.querySelectorAll('input'); return { name: inputs[0]?.value.trim(), url: inputs[1]?.value.trim() }; }).filter(s => s.name && s.url), hacks: { fly: !!document.getElementById('fly')?.checked, fullbright: !!document.getElementById('fullbright')?.checked, xray: !!document.getElementById('xray')?.checked, nofall: !!document.getElementById('nofall')?.checked, speed: !!document.getElementById('speed')?.checked, killaura: document.getElementById('killaura')?.checked ? parseInt(document.getElementById('killauraRange').value) || 4 : false } }; output.textContent = '✅ Settings OK → Fetching /base.html...'; // 2. FETCH ORIGINAL (UNMODIFIED) const response = await fetch('/base.html'); if (!response.ok) throw new Error('❌ /public/base.html missing! Download from eaglercraft.com/paper/download'); let originalHTML = await response.text(); const doc = new DOMParser().parseFromString(originalHTML, 'text/html'); // 3. CREATE MODDED COPY (base stays same) const modScript = doc.createElement('script'); modScript.textContent = ` // MODDED VERSION - Original base.html untouched window.customEagler = ${JSON.stringify(settings)}; console.log('🎮 MODDED CLIENT:', window.customEagler); // Auto-apply hacks after load setTimeout(() => { console.log('🔥 Hacks active:', window.customEagler.hacks); document.title = '${settings.title} - ${settings.splash}'; document.body.style.backgroundColor = '${settings.bgColor}'; if (window.customEagler.hacks.fly) { console.log('✈️ Fly hack loaded'); } if (window.customEagler.hacks.xray) { document.body.classList.add('xray-mode'); } }, 1500); `; doc.head.appendChild(modScript); // Patch title/MOTD visibly doc.title = `${settings.title} - ${settings.splash}`; const motdElements = doc.querySelectorAll('h1, .motd, .title'); motdElements.forEach(el => el.textContent = settings.splash); // 4. GENERATE MODDED HTML const moddedHTML = '\n' + doc.documentElement.outerHTML; // 5. DOWNLOAD NEW FILE (base.html stays original) const blob = new Blob([moddedHTML], { type: 'text/html;charset=utf-8' }); const url = URL.createObjectURL(blob); const link = document.createElement('a'); link.href = url; link.download = `${clientName}_modded.html`; link.click(); document.body.appendChild(link); URL.revokeObjectURL(url); document.body.removeChild(link); // 6. SUCCESS const hackList = getEnabledHacks().join(', ') || 'None'; output.innerHTML = `
✅ ${clientName}_modded.html DOWNLOADED!
✓ Base /base.html UNCHANGED
✓ New file has your hacks/splash/servers
✓ Size: ${(moddedHTML.length/1e6).toFixed(1)}MB
✓ Hacks: ${hackList}
✓ Servers: ${settings.servers.length}
🎮 Open downloaded file → F12 → "MODDED CLIENT" = WORKS
`; // Stats const count = parseInt(localStorage.clientCount || 0) + 1; localStorage.clientCount = count; const countEl = document.getElementById('clientCount'); if (countEl) countEl.textContent = count; } catch (error) { output.innerHTML = `
❌ ${error.message}
💡 Download eaglercraft WASM → save index.html as /public/base.html
`; } } })();
